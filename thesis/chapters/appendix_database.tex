\chapter{Database Schema}

This appendix contains the complete database schema.

\section{Prisma Schema}

\begin{lstlisting}[caption=Complete Prisma Schema]
// prisma/schema.prisma

generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "sqlite"
  url      = env("DATABASE_URL")
}

// User model
model User {
  id            Int             @id @default(autoincrement())
  name          String
  email         String          @unique
  passwordHash  String
  role          String          @default("user")
  createdAt     DateTime        @default(now())
  
  healthProfile HealthProfile?
  mealLogs      MealLog[]
  exerciseLogs  ExerciseLog[]
  dietPlans     DietPlan[]
  workoutPlans  WorkoutPlan[]
  goals         Goal[]
  waterIntake   WaterIntake[]
  favorites     Favorite[]
}

// HealthProfile model
model HealthProfile {
  id            Int      @id @default(autoincrement())
  userId        Int      @unique
  age           Int?
  weight        Float?
  height        Float?
  gender        String?  @default("male")
  activityLevel String?  @default("moderate")
  healthGoals   String?  @default("maintain")
  diseases      String?  // JSON array
  allergens     String?  // JSON array
  createdAt     DateTime @default(now())
  updatedAt     DateTime @default(now()) @updatedAt
  
  user          User     @relation(fields: [userId], 
                         references: [id], onDelete: Cascade)
}

// Goal model
model Goal {
  id              Int      @id @default(autoincrement())
  userId          Int
  goalType        String
  targetValue     Float
  currentValue    Float?
  startValue      Float?
  deadline        DateTime?
  status          String   @default("active")
  createdAt       DateTime @default(now())
  updatedAt       DateTime @updatedAt
  
  user            User     @relation(fields: [userId], 
                           references: [id], onDelete: Cascade)
  
  @@index([userId, status])
}

// DietPlan model
model DietPlan {
  id        Int      @id @default(autoincrement())
  userId    Int
  meals     String   // JSON
  calories  Int?
  createdAt DateTime @default(now())
  
  user      User     @relation(fields: [userId], 
                     references: [id], onDelete: Cascade)
}

// WorkoutPlan model
model WorkoutPlan {
  id        Int      @id @default(autoincrement())
  userId    Int
  exercises String   // JSON
  intensity String?
  createdAt DateTime @default(now())
  
  user      User     @relation(fields: [userId], 
                     references: [id], onDelete: Cascade)
}

// MealLog model
model MealLog {
  id        Int      @id @default(autoincrement())
  userId    Int
  date      DateTime @default(now())
  meals     String   // JSON
  createdAt DateTime @default(now())
  
  user      User     @relation(fields: [userId], 
                     references: [id], onDelete: Cascade)
  
  @@index([userId, date])
}

// ExerciseLog model
model ExerciseLog {
  id        Int      @id @default(autoincrement())
  userId    Int
  date      DateTime @default(now())
  exercises String   // JSON
  createdAt DateTime @default(now())
  
  user      User     @relation(fields: [userId], 
                     references: [id], onDelete: Cascade)
  
  @@index([userId, date])
}

// WaterIntake model
model WaterIntake {
  id        Int      @id @default(autoincrement())
  userId    Int
  date      DateTime @default(now())
  glasses   Int      @default(0)
  goal      Int      @default(8)
  createdAt DateTime @default(now())
  
  user      User     @relation(fields: [userId], 
                     references: [id], onDelete: Cascade)
  
  @@index([userId, date])
  @@unique([userId, date])
}

// Favorite model
model Favorite {
  id        Int      @id @default(autoincrement())
  userId    Int
  itemType  String
  itemId    Int
  itemName  String
  itemData  String   // JSON
  createdAt DateTime @default(now())
  
  user      User     @relation(fields: [userId], 
                     references: [id], onDelete: Cascade)
  
  @@index([userId, itemType])
  @@unique([userId, itemType, itemId])
}

// FoodItem model
model FoodItem {
  id        Int      @id @default(autoincrement())
  name      String
  calories  Int
  allergens String?  // JSON
  createdAt DateTime @default(now())
}

// ExerciseItem model
model ExerciseItem {
  id           Int      @id @default(autoincrement())
  name         String
  impactLevel  String
  createdAt    DateTime @default(now())
}
\end{lstlisting}
