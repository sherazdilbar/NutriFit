// app/api/auth/register/route.ts
export async function POST(request: Request) {
  const { name, email, password } = await request.json();
  
  // Validate input
  if (!name || !email || !password) {
    return NextResponse.json(
      { error: 'All fields required' }, 
      { status: 400 }
    );
  }
  
  // Check existing user
  const existingUser = await prisma.user.findUnique({
    where: { email }
  });
  
  if (existingUser) {
    return NextResponse.json(
      { error: 'Email already registered' }, 
      { status: 400 }
    );
  }
  
  // Hash password and create user
  const passwordHash = await bcrypt.hash(password, 10);
  const user = await prisma.user.create({
    data: { name, email, passwordHash }
  });
  
  // Generate JWT token
  const token = await generateToken({
    userId: user.userId,
    email: user.email
  });
  
  return NextResponse.json({ token, user });
}
\end{lstlisting}

\begin{lstlisting}[language=JavaScript, caption=BMI and Calorie Calculations]
// lib/safetyCheck.ts

// BMI Calculation
export function calculateBMI(weight, height) {
  if (!weight || !height) {
    return { bmi: null, error: 'Missing data' };
  }
  
  const heightInMeters = height / 100;
  const bmi = weight / (heightInMeters * heightInMeters);
  
  let category;
  if (bmi < 18.5) category = 'Underweight';
  else if (bmi < 25) category = 'Normal';
  else if (bmi < 30) category = 'Overweight';
  else category = 'Obese';
  
  return { bmi: Math.round(bmi * 10) / 10, category };
}

// Daily Calorie Calculation (Harris-Benedict)
export function calculateDailyCalories(
  weight, height, age, gender, activityLevel, healthGoals
) {
  let bmr;
  if (gender === 'male') {
    bmr = 88.362 + (13.397 * weight) + 
          (4.799 * height) - (5.677 * age);
  } else {
    bmr = 447.593 + (9.247 * weight) + 
          (3.098 * height) - (4.330 * age);
  }
  
  const multipliers = {
    sedentary: 1.2, light: 1.375, moderate: 1.55,
    active: 1.725, very_active: 1.9
  };
  
  let tdee = bmr * multipliers[activityLevel];
  
  if (healthGoals === 'lose_weight') tdee -= 500;
  else if (healthGoals === 'gain_weight') tdee += 500;
  
  return Math.round(tdee);
}

\section{Safety Checks}

\begin{lstlisting}[language=JavaScript, caption=Exercise Safety Validation]
// lib/safetyCheck.ts
export function checkExerciseSafety(impactLevel, diseases) {
  const highRisk = ['Heart Disease', 'Hypertension'];
  const jointIssues = ['Arthritis', 'Knee Problems', 
                       'Back Pain'];
  
  if (diseases.some(d => highRisk.includes(d)) && 
      impactLevel === 'High') {
    return { 
      safe: false, 
      warning: 'High-intensity not recommended' 
    };
  }
  
  if (diseases.some(d => jointIssues.includes(d)) && 
      impactLevel === 'High') {
    return { 
      safe: false, 
      warning: 'High-impact may worsen joint pain' 
    };
  }
  
  return { safe: true };
}

// lib/planGenerator.ts
export async function generateDietPlan(userId, profile) {
  // Calculate daily calories
  const dailyCalories = calculateDailyCalories(
    profile.weight, profile.height, profile.age,
    profile.gender, profile.activityLevel, 
    profile.healthGoals
  );
  
  // Distribute calories across meals
  const distribution = {
    breakfast: dailyCalories * 0.25,
    lunch: dailyCalories * 0.35,
    dinner: dailyCalories * 0.30,
    snacks: dailyCalories * 0.10
  };
  
  // Get safe foods (exclude allergens)
  const allergens = JSON.parse(profile.allergens || '[]');
  const allFoods = await prisma.foodItem.findMany();
  const safeFoods = allFoods.filter(food => {
    const foodAllergens = JSON.parse(food.allergens || '[]');
    return !foodAllergens.some(a => allergens.includes(a));
  });
  
  // Generate balanced meals
  const meals = {
    breakfast: generateMeal(safeFoods, 'breakfast'),
    lunch: generateMeal(safeFoods, 'lunch'),
    dinner: generateMeal(safeFoods, 'dinner'),
    snacks: generateMeal(safeFoods, 'snacks')
  };
  
  // Save to database
  return await prisma.dietPlan.create({
    data: { 
      userId, 
      meals: JSON.stringify(meals), 
      calories: dailyCalories 
    }
  });
}

\section{Streak Calculation}

\begin{lstlisting}[language=JavaScript, caption=Streak Tracking Algorithm]
// app/api/progress/streaks/route.ts
function calculateStreak(logs) {
  if (logs.length === 0) return 0;
  
  let streak = 0;
  let currentDate = new Date();
  currentDate.setHours(0, 0, 0, 0);
  
  for (const log of logs) {
    const logDate = new Date(log.date);
    logDate.setHours(0, 0, 0, 0);
    
    const daysDiff = Math.floor(
      (currentDate.getTime() - logDate.getTime()) / 
      (1000 * 60 * 60 * 24)
    );
    
    if (daysDiff === streak) {
      streak++;
    } else if (daysDiff > streak) {
      break;
    }
  }
  
  return streak;
}
