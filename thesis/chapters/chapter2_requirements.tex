\chapter{Requirements Analysis}

\section{Functional Requirements}

\subsection{User Management}

\begin{itemize}
    \item \textbf{FR1.1}: Users shall be able to register with name, email, and password
    \item \textbf{FR1.2}: Users shall be able to log in with email and password
    \item \textbf{FR1.3}: The system shall authenticate users using JWT tokens
    \item \textbf{FR1.4}: Users shall be able to log out from the application
    \item \textbf{FR1.5}: Passwords shall be securely hashed using bcrypt
\end{itemize}

\subsection{Health Profile Management}

\begin{itemize}
    \item \textbf{FR2.1}: Users shall be able to create a health profile with age, weight, height, gender
    \item \textbf{FR2.2}: Users shall be able to specify their activity level (sedentary, light, moderate, active, very active)
    \item \textbf{FR2.3}: Users shall be able to set health goals (lose weight, maintain, gain weight)
    \item \textbf{FR2.4}: Users shall be able to list their medical conditions (diabetes, heart disease, asthma, etc.)
    \item \textbf{FR2.5}: Users shall be able to specify food allergens
    \item \textbf{FR2.6}: Users shall be able to view and update their health profile
    \item \textbf{FR2.7}: The system shall calculate BMI based on height and weight
\end{itemize}

\subsection{Diet Planning}

\begin{itemize}
    \item \textbf{FR3.1}: The system shall generate personalized diet plans based on user health profile
    \item \textbf{FR3.2}: Diet plans shall include breakfast, lunch, dinner, and snacks
    \item \textbf{FR3.3}: The system shall calculate total daily calories
    \item \textbf{FR3.4}: The system shall ensure balanced meals with protein, carbohydrates, and vegetables
    \item \textbf{FR3.5}: The system shall exclude foods containing user's allergens
    \item \textbf{FR3.6}: Users shall be able to view their diet plan history
    \item \textbf{FR3.7}: Users shall be able to view individual diet plan details
\end{itemize}

\subsection{Workout Planning}

\begin{itemize}
    \item \textbf{FR4.1}: The system shall generate personalized workout plans based on user health profile
    \item \textbf{FR4.2}: The system shall recommend exercises based on user's medical conditions
    \item \textbf{FR4.3}: The system shall exclude high-impact exercises for users with joint problems
    \item \textbf{FR4.4}: Workout plans shall include exercise name, sets, reps, and rest time
    \item \textbf{FR4.5}: Users shall be able to view their workout plan history
    \item \textbf{FR4.6}: Users shall be able to view individual workout plan details
\end{itemize}

\subsection{Meal Logging}

\begin{itemize}
    \item \textbf{FR5.1}: Users shall be able to log meals with food items and quantities
    \item \textbf{FR5.2}: The system shall calculate total calories for logged meals
    \item \textbf{FR5.3}: The system shall validate that meal dates are not in the future
    \item \textbf{FR5.4}: The system shall prevent duplicate meal logs within 1 hour
    \item \textbf{FR5.5}: Users shall be able to view their meal log history
    \item \textbf{FR5.6}: Users shall be able to delete meal logs
\end{itemize}

\subsection{Exercise Logging}

\begin{itemize}
    \item \textbf{FR6.1}: Users shall be able to log exercises with duration and intensity
    \item \textbf{FR6.2}: The system shall validate that exercise dates are not in the future
    \item \textbf{FR6.3}: The system shall prevent duplicate exercise logs within 2 hours
    \item \textbf{FR6.4}: Users shall be able to view their exercise log history
    \item \textbf{FR6.5}: Users shall be able to delete exercise logs
\end{itemize}

\subsection{Goal Management}

\begin{itemize}
    \item \textbf{FR7.1}: Users shall be able to create goals (weight, calories, workouts)
    \item \textbf{FR7.2}: Users shall be able to set target values and deadlines for goals
    \item \textbf{FR7.3}: The system shall track current progress towards goals
    \item \textbf{FR7.4}: Users shall be able to view goal progress with visual indicators
    \item \textbf{FR7.5}: Users shall be able to update goal status (active, completed, cancelled)
    \item \textbf{FR7.6}: Users shall be able to delete goals
\end{itemize}

\subsection{Food and Exercise Catalogs}

\begin{itemize}
    \item \textbf{FR8.1}: Users shall be able to browse food catalog by category
    \item \textbf{FR8.2}: Users shall be able to search foods by name
    \item \textbf{FR8.3}: The system shall display allergen warnings for unsafe foods
    \item \textbf{FR8.4}: Users shall be able to filter foods by calorie range
    \item \textbf{FR8.5}: Users shall be able to browse exercise catalog by category
    \item \textbf{FR8.6}: Users shall be able to search exercises by name
    \item \textbf{FR8.7}: The system shall display safety warnings for exercises based on medical conditions
    \item \textbf{FR8.8}: Users shall be able to filter exercises by impact level
\end{itemize}

\subsection{Progress Tracking}

\begin{itemize}
    \item \textbf{FR9.1}: The system shall calculate weekly summary (meals, workouts, calories, active days)
    \item \textbf{FR9.2}: The system shall track meal logging streaks
    \item \textbf{FR9.3}: The system shall track exercise logging streaks
    \item \textbf{FR9.4}: The system shall award badges for 7-day and 30-day streaks
    \item \textbf{FR9.5}: Users shall be able to view progress statistics and charts
\end{itemize}

\subsection{Water Intake Tracking}

\begin{itemize}
    \item \textbf{FR10.1}: Users shall be able to track daily water intake in glasses
    \item \textbf{FR10.2}: The system shall set a default goal of 8 glasses per day
    \item \textbf{FR10.3}: Users shall be able to add or remove glasses
    \item \textbf{FR10.4}: The system shall display progress towards daily water goal
\end{itemize}

\subsection{Favorites System}

\begin{itemize}
    \item \textbf{FR11.1}: Users shall be able to mark foods as favorites
    \item \textbf{FR11.2}: Users shall be able to mark exercises as favorites
    \item \textbf{FR11.3}: Users shall be able to view their favorite items
    \item \textbf{FR11.4}: Users shall be able to remove items from favorites
\end{itemize}

\subsection{Reminders}

\begin{itemize}
    \item \textbf{FR12.1}: The system shall remind users if no meal is logged today
    \item \textbf{FR12.2}: The system shall remind users if no exercise in last 3 days
    \item \textbf{FR12.3}: The system shall remind users if water intake is below goal
    \item \textbf{FR12.4}: The system shall remind users about goals with approaching deadlines
\end{itemize}

\subsection{Recipe Suggestions}

\begin{itemize}
    \item \textbf{FR13.1}: The system shall generate recipe suggestions from food catalog
    \item \textbf{FR13.2}: Recipes shall combine protein, carbohydrates, and vegetables
    \item \textbf{FR13.3}: The system shall calculate total calories for recipes
    \item \textbf{FR13.4}: The system shall exclude foods with user allergens from recipes
\end{itemize}

\section{Non-Functional Requirements}

\subsection{Performance}

\begin{itemize}
    \item \textbf{NFR1.1}: Page load time shall not exceed 3 seconds
    \item \textbf{NFR1.2}: API response time shall not exceed 1 second
    \item \textbf{NFR1.3}: The application shall support at least 100 concurrent users
\end{itemize}

\subsection{Security}

\begin{itemize}
    \item \textbf{NFR2.1}: All passwords shall be hashed using bcrypt
    \item \textbf{NFR2.2}: Authentication shall use JWT tokens
    \item \textbf{NFR2.3}: API endpoints shall validate user authentication
    \item \textbf{NFR2.4}: Sensitive data shall not be exposed in client-side code
\end{itemize}

\subsection{Usability}

\begin{itemize}
    \item \textbf{NFR3.1}: The user interface shall be intuitive and easy to navigate
    \item \textbf{NFR3.2}: The application shall be responsive on mobile, tablet, and desktop
    \item \textbf{NFR3.3}: Error messages shall be clear and helpful
    \item \textbf{NFR3.4}: The application shall provide visual feedback for user actions
\end{itemize}

\subsection{Reliability}

\begin{itemize}
    \item \textbf{NFR4.1}: The application shall have 99\% uptime
    \item \textbf{NFR4.2}: Data shall be persisted reliably in the database
    \item \textbf{NFR4.3}: The application shall handle errors gracefully
\end{itemize}

\subsection{Maintainability}

\begin{itemize}
    \item \textbf{NFR5.1}: Code shall follow TypeScript best practices
    \item \textbf{NFR5.2}: Code shall be well-documented
    \item \textbf{NFR5.3}: The application shall use a modular architecture
\end{itemize}

\section{User Stories}

\begin{itemize}
    \item As a user, I want to create an account so that I can access the application
    \item As a user, I want to create a health profile so that I receive personalized recommendations
    \item As a user, I want to generate a diet plan so that I know what to eat
    \item As a user, I want to log my meals so that I can track my nutrition
    \item As a user, I want to log my exercises so that I can track my fitness
    \item As a user, I want to set goals so that I can monitor my progress
    \item As a user, I want to see my weekly summary so that I understand my habits
    \item As a user, I want to track my water intake so that I stay hydrated
    \item As a user, I want to save favorite foods so that I can quickly log them
    \item As a user, I want to receive reminders so that I stay consistent
\end{itemize}
