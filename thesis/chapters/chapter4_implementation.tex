\subsection{Core Authentication}

\begin{lstlisting}[language=JavaScript, caption=JWT Token Generation]
// lib/tokenHandler.ts
import { SignJWT, jwtVerify } from 'jose';

const secret = new TextEncoder().encode(
  process.env.JWT_SECRET
);

export async function generateToken(payload: any) {
  return await new SignJWT(payload)
    .setProtectedHeader({ alg: 'HS256' })
    .setExpirationTime('7d')
    .sign(secret);
}

export async function verifyToken(token: string) {
  const { payload } = await jwtVerify(token, secret);
  return payload;
}
\end{lstlisting}

\subsection{Health Calculations}

\subsubsection{BMI Calculation}

\begin{lstlisting}[language=JavaScript, caption=BMI Calculator]
export function calculateBMI(weight: number, height: number) {
  if (!weight || !height) {
    return { bmi: null, error: 'Missing data' };
  }
  
  const heightInMeters = height / 100;
  const bmi = weight / (heightInMeters * heightInMeters);
  
  let category;
  if (bmi < 18.5) category = 'Underweight';
  else if (bmi < 25) category = 'Normal';
  else if (bmi < 30) category = 'Overweight';
  else category = 'Obese';
  
  return { bmi: Math.round(bmi * 10) / 10, category };
}
\end{lstlisting}

\subsubsection{Daily Calorie Calculation}

\begin{lstlisting}[language=JavaScript, caption=Calorie Calculator]
export function calculateDailyCalories(
  weight, height, age, gender, activityLevel, healthGoals
) {
  // Harris-Benedict Equation
  let bmr;
  if (gender === 'male') {
    bmr = 88.362 + (13.397 * weight) + 
          (4.799 * height) - (5.677 * age);
  } else {
    bmr = 447.593 + (9.247 * weight) + 
          (3.098 * height) - (4.330 * age);
  }
  
  // Activity multipliers
  const multipliers = {
    sedentary: 1.2, light: 1.375, moderate: 1.55,
    active: 1.725, very_active: 1.9
  };
  
  let tdee = bmr * multipliers[activityLevel];
  
  // Adjust for goals
  if (healthGoals === 'lose_weight') tdee -= 500;
  else if (healthGoals === 'gain_weight') tdee += 500;
  
  return Math.round(tdee);
}
\end{lstlisting}

\subsection{Diet Plan Generation}

export async function generateDietPlan(userId, profile) {
  // Calculate daily calories
  const dailyCalories = calculateDailyCalories(
    profile.weight, profile.height, profile.age,
    profile.gender, profile.activityLevel, 
    profile.healthGoals
  );
  
  // Distribute calories
  const distribution = {
    breakfast: dailyCalories * 0.25,
    lunch: dailyCalories * 0.35,
    dinner: dailyCalories * 0.30,
    snacks: dailyCalories * 0.10
  };
  
  // Get safe foods (exclude allergens)
  const allergens = JSON.parse(profile.allergens || '[]');
  const allFoods = await prisma.foodItem.findMany();
  const safeFoods = allFoods.filter(food => {
    const foodAllergens = JSON.parse(food.allergens || '[]');
    return !foodAllergens.some(a => allergens.includes(a));
  });
  
  // Generate balanced meals
  const meals = {
    breakfast: generateMeal(safeFoods, 'breakfast'),
    lunch: generateMeal(safeFoods, 'lunch'),
    dinner: generateMeal(safeFoods, 'dinner'),
    snacks: generateMeal(safeFoods, 'snacks')
  };
  
  // Save to database
  return await prisma.dietPlan.create({
    data: { userId, meals: JSON.stringify(meals), 
            calories: dailyCalories }
  });
}

\subsection{Safety Checks}

\begin{lstlisting}[language=JavaScript, caption=Exercise Safety Check]
export function checkExerciseSafety(
  impactLevel, diseases
) {
  const highRisk = ['Heart Disease', 'Hypertension'];
  const jointIssues = ['Arthritis', 'Knee Problems', 
                       'Back Pain'];
  
  if (diseases.some(d => highRisk.includes(d)) && 
      impactLevel === 'High') {
    return { 
      safe: false, 
      warning: 'High-intensity not recommended' 
    };
  }
  
  if (diseases.some(d => jointIssues.includes(d)) && 
      impactLevel === 'High') {
    return { 
      safe: false, 
      warning: 'High-impact may worsen joint pain' 
    };
  }
  
  return { safe: true };
}
\end{lstlisting}

\subsection{API Communication}

\begin{lstlisting}[language=JavaScript, caption=API Client Helper]
export async function apiRequest(endpoint, options = {}) {
  const token = localStorage.getItem('token');
  
  const response = await fetch(endpoint, {
    ...options,
    headers: {
      'Content-Type': 'application/json',
      Authorization: `Bearer ${token}`,
      ...options.headers
    }
  });
  
  if (!response.ok) {
    const error = await response.json();
    throw new Error(error.error || 'Request failed');
  }
  
  return response.json();
}
\end{lstlisting}
